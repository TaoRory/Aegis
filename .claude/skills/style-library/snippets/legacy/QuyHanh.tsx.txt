/** Khối điêu khắc cẩm thạch: raymarching một mặt cầu biến dạng mềm, có vân đá, xoay theo chuột */
const SCULPTURE = `#version 300 es
precision highp float;
uniform float uTime; uniform vec2 uRes; uniform vec2 uMouse; uniform float uScroll;
out vec4 outColor;
mat2 rot(float a){ float c = cos(a), s = sin(a); return mat2(c, -s, s, c); }
float map(vec3 p){
  p.xz *= rot(uTime * 0.12 + (uMouse.x - 0.5) * 1.4 + uScroll * 2.0);
  p.yz *= rot((uMouse.y - 0.5) * 0.9);
  float d = length(p) - 1.0;
  d += 0.16 * sin(p.x * 2.3 + uTime * 0.5) * sin(p.y * 2.7 + uTime * 0.35) * sin(p.z * 2.1 + 1.3);
  d -= 0.05 * sin(p.y * 9.0 + p.x * 3.0);
  return d * 0.6;
}
vec3 calcNormal(vec3 p){
  vec2 e = vec2(0.0015, 0.0);
  return normalize(vec3(map(p + e.xyy) - map(p - e.xyy), map(p + e.yxy) - map(p - e.yxy), map(p + e.yyx) - map(p - e.yyx)));
}
void main(){
  vec2 uv = (gl_FragCoord.xy - 0.5 * uRes) / uRes.y;
  vec3 ro = vec3(0.0, 0.0, 3.3);
  vec3 rd = normalize(vec3(uv, -1.55));
  float t = 0.0; bool hit = false;
  for (int i = 0; i < 90; i++) {
    float d = map(ro + rd * t);
    if (d < 0.0008) { hit = true; break; }
    t += d;
    if (t > 7.0) break;
  }
  if (!hit) { outColor = vec4(0.0); return; }
  vec3 p = ro + rd * t;
  vec3 n = calcNormal(p);
  vec3 l = normalize(vec3(-0.5, 0.85, 0.55));
  float diff = clamp(dot(n, l), 0.0, 1.0);
  float amb = 0.55 + 0.45 * n.y;
  float fres = pow(1.0 - max(dot(n, -rd), 0.0), 3.0);
  float spec = pow(max(dot(reflect(-l, n), -rd), 0.0), 40.0);
  float vein = smoothstep(0.9, 1.0, sin((p.x + p.y * 0.7) * 6.5 + sin(p.z * 4.0) * 1.8) * 0.5 + 0.5);
  vec3 stone = vec3(0.93, 0.93, 0.91);
  vec3 shade = vec3(0.66, 0.70, 0.69);
  vec3 col = mix(shade, stone, diff * 0.65 + amb * 0.35);
  col = mix(col, vec3(0.16, 0.32, 0.29), vein * 0.28);
  col += spec * 0.22 + fres * vec3(0.10, 0.14, 0.13);
  outColor = vec4(col, 1.0);
}`;

export function QuyHanh() {
  const root = useRef<HTMLDivElement>(null);

  useGSAP(
    () => {
      const mm = gsap.matchMedia();
      mm.add(MQ.anyMotion, () => {
        // Số liệu đếm lên khi vào khung nhìn
        root.current?.querySelectorAll<HTMLElement>("[data-count]").forEach((el) => {
          const target = Number(el.dataset.count);
          const obj = { v: 0 };
          gsap.to(obj, {
            v: target,
            duration: 1.8,
            ease: "power3.out",
            onUpdate: () => (el.textContent = String(Math.round(obj.v))),
            scrollTrigger: { trigger: el, start: "top 85%", once: true },
          });
        });
        // Biểu đồ tự vẽ theo cuộn
        const line = root.current?.querySelector<SVGPathElement>("[data-chart-line]");
        const area = root.current?.querySelector<SVGPathElement>("[data-chart-area]");
        if (line && area) {
          const length = line.getTotalLength();
          gsap.set(line, { strokeDasharray: length, strokeDashoffset: length });
          const tl = gsap.timeline({ scrollTrigger: { trigger: line, start: "top 80%", end: "bottom 40%", scrub: 1 } });
          tl.to(line, { strokeDashoffset: 0, ease: "none" }).from(area, { opacity: 0, ease: "none" }, 0.3);
        }
      });
      return () => mm.revert();
    },
    { scope: root },
  );

  return (
    <div ref={root} className="tpl tpl-quy-hanh">
      <header className="absolute inset-x-0 top-0 z-20">
        <div className="container-page flex h-20 items-center justify-between">
          <a href="#dau-trang" className="flex items-center gap-2 text-lg font-medium">
            <span aria-hidden className="size-4 rounded-full border-[3px] border-(--t-accent)" />
            Quỹ Hành
          </a>
          <nav aria-label="Điều hướng Quỹ Hành" className="hidden md:block">
            <ul className="flex gap-8 text-sm text-(--t-muted)">
              {[
                ["Chúng tôi làm gì", "#lam-gi"],
                ["Nguyên tắc", "#nguyen-tac"],
                ["Góc nhìn", "#goc-nhin"],
              ].map(([label, href]) => (
                <li key={href}>
                  <a href={href} className="hover:text-(--t-fg)">
                    {label}
                  </a>
                </li>
              ))}
            </ul>
          </nav>
          <a href="#demo-ket" className="quy-hanh-pill">
            Đăng nhập
          </a>
        </div>
      </header>

      {/* Hero với khối điêu khắc */}
      <section id="dau-trang" aria-labelledby="quy-hanh-title" className="relative flex min-h-svh flex-col items-center overflow-hidden pt-24 text-center">
        <div aria-hidden className="quy-hanh-halo absolute top-[8%] left-1/2 size-[70vmin] -translate-x-1/2 rounded-full" />
        <div aria-hidden className="relative mt-4 size-[min(56vmin,30rem)]">
          <ShaderCanvas fragment={SCULPTURE} maxDpr={1.25} />
        </div>
        <SplitReveal as="h1" id="quy-hanh-title" trigger="intro" className="relative mt-2 text-[clamp(2.6rem,1rem+4.8vw,6rem)] leading-[1.02] tracking-[-0.035em]">
          <span className="block">Đầu tư đúng lúc,</span>
          <span className="block">
            với <em>góc nhìn dài hạn</em>
          </span>
        </SplitReveal>
        <p className="relative mt-6 max-w-[34rem] px-6 leading-relaxed text-(--t-muted)">
          Quỹ Hành là đơn vị quản lý tài sản theo hướng dữ liệu, giúp nhà đầu tư cá nhân tiếp cận chiến lược của tổ chức.
        </p>
        <a href="#lam-gi" className="quy-hanh-pill relative mt-8 mb-16">
          Tìm hiểu thêm
        </a>
      </section>

      {/* Chúng tôi làm gì + số liệu */}

                    {stat.suffix}

            ))}

        </div>
      </section>

      {/* Góc nhìn */}
      <section id="goc-nhin" aria-labelledby="quy-hanh-insights" className="pb-32">
        <div className="container-page">
          <h2 id="quy-hanh-insights" className="text-[clamp(2.2rem,1rem+3vw,4rem)] leading-[1.05] tracking-[-0.03em]">
            Góc <em>nhìn</em>
          </h2>
          <ul className="mt-12 grid gap-4 md:grid-cols-3">
            {insights.map((item) => (
              <li key={item.title} className="quy-hanh-stat flex min-h-56 flex-col justify-between p-7 transition-colors hover:bg-(--t-surface)">
                <span className="text-xs text-(--t-muted)">{item.date}</span>
                <span className="text-xl leading-snug tracking-[-0.02em]">{item.title}</span>
              </li>
            ))}
          </ul>
        </div>
      </section>
    </div>
  );
