function Countdown() {
  const [now, setNow] = useState<number | null>(null);
  useEffect(() => {
    const id = window.setInterval(() => setNow(Date.now()), 1000);
    const first = window.setTimeout(() => setNow(Date.now()), 0);
    return () => {
      window.clearInterval(id);
      window.clearTimeout(first);
    };
  }, []);
  const left = now === null ? null : Math.max(0, EVENT_TIME - now);
  const parts = [
    ["Ngày", left === null ? null : Math.floor(left / 86_400_000)],
    ["Giờ", left === null ? null : Math.floor(left / 3_600_000) % 24],
    ["Phút", left === null ? null : Math.floor(left / 60_000) % 60],
    ["Giây", left === null ? null : Math.floor(left / 1000) % 60],
  ] as const;
  return (
    <dl aria-label="Đếm ngược tới ngày diễn ra" className="grid grid-cols-4 gap-2">
      {parts.map(([label, value]) => (
        <div key={label} className="border border-(--t-line) px-3 py-2">
          <dt className="ddt-label text-(--t-muted)">{label}</dt>
          <dd className="ddt-display mt-1 text-[clamp(1.8rem,1rem+2vw,3rem)] tabular-nums">
            {value === null ? "--" : String(value).padStart(2, "0")}
          </dd>
        </div>
      ))}
    </dl>
  );
}

export function DemDoThi() {

  const [day, setDay] = useState(0);
  const tabsRef = useRef<HTMLDivElement>(null);

  useGSAP(
    () => {
      const mm = gsap.matchMedia();
      mm.add(MQ.anyMotion, () => {
        const lines = heroRef.current?.querySelectorAll<HTMLElement>("[data-line]") ?? [];
        const inners = heroRef.current?.querySelectorAll<HTMLElement>("[data-line-inner]") ?? [];
        // Vào trang (lớp trong): từng dòng trượt vào từ hai phía
        gsap.from(inners, { xPercent: (i) => (i % 2 ? 40 : -40), opacity: 0, duration: 1.2, ease: "expo.out", stagger: 0.1 });

        lines.forEach((line, i) => {
          gsap.to(line, {
            xPercent: i % 2 ? -18 : 18,
            ease: "none",
            scrollTrigger: { trigger: heroRef.current, start: "top top", end: "bottom top", scrub: true },
          });
        });
      });
      return () => mm.revert();
    },
    { scope: heroRef },
  );

  const onTabKey = (event: KeyboardEvent<HTMLButtonElement>) => {
    if (event.key !== "ArrowRight" && event.key !== "ArrowLeft") return;
